import { Card, CardGrid } from "@astrojs/starlight/components";

```ts
import { $ } from "#mc";

/** @mcfunction */
export function StartRound() {
  $.Tp("@a", "0 ~10 0");
  $.Tellraw("@a", { text: "Fight!", color: "red" });
  $.Execute.As("@a").At("@s").Run.Say("go!");
  $.Function(Cleanup);
}

function Cleanup() {
  $.Say("cleaning up");
}
```
